\section*{前言：三步走完整篇論文}
\addcontentsline{toc}{section}{前言：三步走完整篇論文}

整篇論文的架構可以用三個等號串起來：

\[
  \underbrace{\VIX_T^2 = aV_T + b}_{\text{Step 1（eq 5）}}
  \;\xrightarrow{\text{Step 2：}\sqrt{x}\text{ 的 Laplace 技巧}}\;
  \E^\Q_t[\VIX_T] = \int_0^\infty(\cdots)\,\varphi(-au)\,\dd u
  \;\xrightarrow{\text{Step 3：affine MGF}}\;
  \varphi = e^{C+DV_t+A}
\]

\begin{itemize}
  \item \textbf{Step 1（Sec 2.2，eq 5）}：在 SVJJ 模型下，VIX 的平方對瞬時方差 $V_t$ 是線性的。
  這個線性結構是整個定價問題可以有解析解的根本原因。
  \item \textbf{Step 2（Proposition 1，eq 8）}：$\E[\sqrt{\text{線性函數}}]$
  可以用 Gaussian 積分恆等式轉換成對 $V_T$ 的 Laplace transform 的積分。
  \item \textbf{Step 3（eq 10--11）}：SVJJ 模型屬於 affine 擴散，$V_T$ 的 MGF 有閉式解
  $e^{C+DV_t}$，所以 Step 2 的積分被積函數是已知的解析函數。
\end{itemize}

最終得到一個\textbf{一維實值積分}——比二維 Fourier 反演快、比 Taylor 近似精確。

在進入 Steps 前，先把數學工具補齊（Section 1）。

\section{隨機微積分基礎}

\subsection{布朗運動與二次變差}

標準 Wiener 過程（布朗運動）$W_t$ 定義為：
\begin{itemize}
  \item $W_0 = 0$
  \item 增量 $W_t - W_s \sim N(0, t-s)$，對所有 $0 \le s < t$
  \item 不重疊時段的增量互相獨立
  \item 路徑連續（但幾乎處處不可微）
\end{itemize}

路徑不可微，所以不能用普通的黎曼積分定義 $\int_0^T f(t)\,\dd W_t$。
Itô 積分的建構思路是把 $[0,T]$ 分成小段，取左端點的 Riemann 和：
\[
  \int_0^T f(t)\,\dd W_t
  \;:=\; \lim_{n\to\infty} \sum_{k} f(t_k)\,(W_{t_{k+1}} - W_{t_k})
\]
這個極限在均方意義下存在（只要 $\int_0^T \E[f(t)^2]\,\dd t < \infty$）。

\subsubsection*{二次變差（Quadratic Variation）}

對普通連續可微路徑 $g(t)$：
$\sum_k (g(t_{k+1})-g(t_k))^2 \to 0$（趨近於 0，因為各項都是 $O((\Delta t)^2)$）。

對 $W_t$：$\sum_k (W_{t_{k+1}}-W_{t_k})^2 \to T$（在 $L^2$ 意義下，等於 $t$），因為
每一項 $\E[(W_{t_{k+1}}-W_{t_k})^2] = \Delta t$，加總為 $T$，而
$\Var(\sum_k(W_{t_{k+1}}-W_{t_k})^2) = \sum_k 2(\Delta t)^2 \to 0$。

這個結果通常寫成\textbf{Itô 乘法規則}：
\begin{equation}
  (\dd W_t)^2 = \dd t,
  \qquad (\dd t)^2 = 0,
  \qquad \dd t\,\dd W_t = 0
  \label{eq:ito-rules}
\end{equation}
以上是 Itô 微積分與普通微積分不同的根本之處。

\subsection{Itô 引理（Itô's Lemma）}

對一般 SDE $\dd X_t = \mu(X_t, t)\,\dd t + \sigma(X_t, t)\,\dd W_t$，以及光滑函數 $f(x,t)$，
Itô 引理給出 $f(X_t, t)$ 的 SDE：

\begin{keybox}
\[
  \dd f(X_t,t)
  = \frac{\partial f}{\partial t}\,\dd t
  + \frac{\partial f}{\partial x}\,\dd X_t
  + \frac{1}{2}\frac{\partial^2 f}{\partial x^2}(\dd X_t)^2
\]
展開 $(\dd X_t)^2 = \sigma^2\,\dd t$（用規則 \eqref{eq:ito-rules}）後：
\begin{equation}
  \dd f = \underbrace{\left(\frac{\partial f}{\partial t}
    + \mu\frac{\partial f}{\partial x}
    + \frac{\sigma^2}{2}\frac{\partial^2 f}{\partial x^2}\right)}_{\text{drift（dt 項）}}\dd t
  + \underbrace{\sigma\frac{\partial f}{\partial x}}_{\text{diffusion（dW 項）}}\dd W_t
  \label{eq:ito-lemma}
\end{equation}
\end{keybox}

\textbf{推導邏輯：}就是 Taylor 展開，但多了 $(\dd W)^2 = \dd t$ 這一項。

\begin{derivbox}
對 $f(X_t)$ 做 Taylor 展開到二階：
\[
  \dd f = f'\,\dd X + \tfrac{1}{2}f''\,(\dd X)^2 + O((\dd t)^{3/2})
\]
代入 $\dd X = \mu\,\dd t + \sigma\,\dd W$，計算 $(\dd X)^2$：
\[
  (\dd X)^2
  = \mu^2(\dd t)^2 + 2\mu\sigma\,\dd t\,\dd W + \sigma^2(\dd W)^2
  = 0 + 0 + \sigma^2\,\dd t
\]
（後兩條規則為 $O((\dd t)^{3/2})$ 和 $O((\dd t)^2)$，忽略。）
代回：$\dd f = f'(\mu\,\dd t + \sigma\,\dd W) + \tfrac{1}{2}f''\sigma^2\,\dd t$
\end{derivbox}

\begin{physicsbox}
\textbf{BS 方程 = 熱擴散方程（Fokker-Planck）}

對 $V(S,t)$（選擇權價值）套用 Itô 引理，加上 no-arbitrage（$\dd V = r V\,\dd t$），得到
Black-Scholes PDE：
\[
  \frac{\partial V}{\partial t}
  + \frac{1}{2}\sigma^2 S^2 \frac{\partial^2 V}{\partial S^2}
  + rS\frac{\partial V}{\partial S} - rV = 0
\]
換變數 $x = \ln S$，$\tau = T-t$：
\[
  \frac{\partial V}{\partial \tau}
  = \underbrace{\frac{1}{2}\sigma^2\frac{\partial^2 V}{\partial x^2}}_{\text{熱擴散}}
  + \underbrace{\left(r - \frac{\sigma^2}{2}\right)\frac{\partial V}{\partial x}}_{\text{對流（drift）}}
  - rV
\]
這是\textbf{對流-擴散方程}，在物理上描述有漂流的溫度場：$\sigma^2/2$ 是熱擴散係數，$r - \sigma^2/2$ 是「風速」。European call 的解析解就是這個 PDE 的 Green's function（基本解的摺積）。

同樣的邏輯適用於 SVJJ 模型——SDE 的結構決定了定價 PDE 的形式，而 affine 結構讓這個 PDE 有解析解。
\end{physicsbox}

\subsection{Feynman-Kac 定理：SDE 與 PDE 的橋}

這是 Section 4 推導 MGF 所需的核心定理。

\begin{keybox}
\textbf{Feynman-Kac（無折現版本）：}

設 $X_t$ 滿足 $\dd X_t = \mu(X_t)\,\dd t + \sigma(X_t)\,\dd W_t$。定義：
\[
  u(x, t) = \E^\Q_t\!\left[f(X_T)\,\Big|\, X_t = x\right]
\]

則 $u$ 滿足\textbf{倒向 Kolmogorov 方程（backward equation）}：
\begin{equation}
  \frac{\partial u}{\partial t}
  + \mu(x)\frac{\partial u}{\partial x}
  + \frac{1}{2}\sigma^2(x)\frac{\partial^2 u}{\partial x^2} = 0,
  \qquad u(x, T) = f(x)
  \label{eq:fk}
\end{equation}
\end{keybox}

\textbf{方向性說明：}$u$ 往\emph{回}對 $t$ 積分（terminal condition at $T$，往 $t < T$ 方向傳播），
與物理的熱方程方向相反——這是「倒向」的意思。

\begin{physicsbox}
\textbf{Feynman path integral 的嚴格版本}

Feynman-Kac 定理是費曼路徑積分（Feynman path integral）的數學嚴格化。

費曼的想法：量子系統從 $x$ 傳播到 $x'$ 的機率幅是對所有可能路徑的加權求和：
\[
  K(x', T; x, t) = \int_{\text{all paths}} e^{iS[\text{path}]/\hbar}\,\mathcal{D}\text{path}
\]
做 Wick 旋轉（$t \to -i\tau$，虛時間）後，$e^{iS/\hbar}$ 變成 $e^{-S_E/\hbar}$（Boltzmann 因子），
路徑積分變成 Wiener 測度（布朗運動的隨機遊走測度）：
\[
  K_E(x', T; x, 0) = \E\!\left[f(X_T)\,\Big|\,X_0=x\right]
\]
這正是 Feynman-Kac。數學金融裡的定價期望值
$\E^\Q[f(S_T)]$ 是費曼路徑積分在實時間的類比。

後面用 Feynman-Kac 推 Riccati ODE 的邏輯，在物理上對應的是：
對作用量為二次型的系統（諧振子、Landau-Zener 等），路徑積分精確等於鞍點展開的結果——
指數仿射的 MGF 是同樣的數學結構。
\end{physicsbox}

\section{SVJJ 模型}

\subsection{模型方程（eq 3--4）}

在風險中性測度 $\Q$ 下，股價 $S_t$ 和瞬時方差 $V_t$ 的 SDE：

\begin{align}
  \frac{\dd S_t}{S_t}
  &= (r - \lambda^\Q\bar\mu^\Q_S)\,\dd t
    + \sqrt{V_t}\,\dd W_t^{\Q,S}
    + \dd J_t^S
  \tag{eq 3}\\[4pt]
  \dd V_t
  &= \kappa^\Q(\theta^\Q - V_t)\,\dd t
    + \sigma_V\sqrt{V_t}\,\dd W_t^{\Q,V}
    + \dd J_t^V
  \tag{eq 4}
\end{align}

其中 $\dd W^{\Q,S}$ 與 $\dd W^{\Q,V}$ 之相關係數為 $\rho$。

跳躍部分：$J^S_t = \sum_{k=1}^{N_t} Z^S_k$，$J^V_t = \sum_{k=1}^{N_t} Z^V_k$，
其中 $N_t$ 是 Poisson 過程（強度 $\lambda^\Q$），$Z^V_k \sim \text{Exp}(\mu_V)$（方差跳躍幅度），
$Z^S_k | Z^V_k \sim N(\mu_S + \rho_J Z^V_k, \sigma_S^2)$（股價跳躍幅度）。
兩者\textbf{同時發生}（simultaneous jumps）。

\textbf{巢狀結構：}$\lambda^\Q = 0$ 退化為純 Heston SV 模型；只有 $J^V = 0$ 退化為 SVJ；只有 $J^S = 0$ 退化為 SVVJ。

\subsection{CIR 過程的條件期望（為 eq 5 鋪墊）}

去掉跳躍項，eq 4 的 diffusion 部分是 CIR 過程：
\begin{equation}
  \dd V_t = \kappa^\Q(\theta^\Q - V_t)\,\dd t + \sigma_V\sqrt{V_t}\,\dd W_t^V
  \label{eq:cir}
\end{equation}

\textbf{條件期望 $\E^\Q_t[V_s]$ 的計算：}設 $g(s) = \E^\Q_t[V_s \,|\, V_t]$。
對 \eqref{eq:cir} 取期望（$\E[\dd W] = 0$）：
\[
  \frac{\dd g}{\dd s} = \kappa^\Q(\theta^\Q - g(s))
\]
這是一階線性 ODE（指數衰減），初始條件 $g(t) = V_t$，解為：
\begin{equation}
  \E^\Q_t[V_s\,|\, V_t]
  = \theta^\Q + (V_t - \theta^\Q)\,e^{-\kappa^\Q(s-t)}
  \label{eq:cir-mean}
\end{equation}

直覺：$V_s$ 以速率 $\kappa^\Q$ 指數收斂到長期均值 $\theta^\Q$，初始偏差 $(V_t - \theta^\Q)$ 不斷縮小。

\section{Step 1：$\VIX_t^2 = aV_t + b$（eq 5--6 的推導）}

\subsection{VIX² 的積分表示}

CBOE 定義：
\begin{equation}
  \VIX_t^2 = -\frac{2}{\bar{T}}\,\E^\Q_t\!\left[\ln\frac{F_{t,\bar{T}}}{S_t}\right]
  \tag{eq 1}
\end{equation}

在 SVJJ 模型下（詳見論文 Sec 2.1 的計算），代入 SDE 並用 Itô 引理化簡：
\begin{equation}
  \VIX_t^2
  = \underbrace{\frac{1}{\bar{T}}\,\E^\Q_t\!\left[\int_t^{t+\bar{T}} V_s\,\dd s\right]}_{\text{連續部分}}
  + \underbrace{\frac{\lambda^\Q}{\bar{T}}\,\E^\Q\!\left[\left(e^{Z^S} - 1 - Z^S\right)\bar{T}\right]}_{\text{跳躍修正（對 } V_t \text{ 是常數）}}
  \label{eq:vix-integral}
\end{equation}

\textbf{跳躍修正項不依賴 $V_t$}（只含跳躍幅度的期望，是模型參數的函數）。因此，線性性質完全由第一項決定。

\subsection{連續部分的精確計算}

\begin{derivbox}
利用 \eqref{eq:cir-mean}，計算積分的期望值：
\begin{align*}
  \E^\Q_t\!\left[\int_t^{t+\bar{T}} V_s\,\dd s\right]
  &= \int_t^{t+\bar{T}} \E^\Q_t[V_s]\,\dd s \\
  &= \int_0^{\bar{T}} \left[\theta^\Q + (V_t - \theta^\Q)\,e^{-\kappa^\Q u}\right]\dd u
  \quad (u = s-t)\\
  &= \theta^\Q\bar{T} + (V_t - \theta^\Q)\,\frac{1 - e^{-\kappa^\Q\bar{T}}}{\kappa^\Q}
\end{align*}
除以 $\bar{T}$，整理成 $V_t$ 的線性形式：
\begin{equation}
  \frac{1}{\bar{T}}\E^\Q_t\!\left[\int_t^{t+\bar{T}}V_s\,\dd s\right]
  = \underbrace{\frac{1-e^{-\kappa^\Q\bar{T}}}{\kappa^\Q\bar{T}}}_{=:\,a_\mathrm{diff}}\,V_t
  + \underbrace{\theta^\Q\!\left(1 - \frac{1-e^{-\kappa^\Q\bar{T}}}{\kappa^\Q\bar{T}}\right)}_{=\,\theta^\Q(1-a_\mathrm{diff})}
  \label{eq:vix2-linear}
\end{equation}
\end{derivbox}

\begin{keybox}
\textbf{eq 5：}結合連續部分 \eqref{eq:vix2-linear} 和常數跳躍修正：
\begin{equation}
  \VIX_t^2 = a\cdot V_t + b
  \tag{eq 5}
\end{equation}
其中（eq 6）：
\[
  a = \frac{1-e^{-\kappa^\Q\bar{T}}}{\kappa^\Q\bar{T}}
  \quad\text{（僅由 CIR diffusion 決定）}
\]
$b$ 含跳躍修正，由 $\theta^\Q$、跳躍參數、$\bar{T}$ 決定（見論文 eq 6 精確表達式）。
\end{keybox}

\textbf{為什麼 $a < 1$？}

$a = (1-e^{-\kappa\bar{T}})/(\kappa\bar{T}) \in (0,1)$，是個「折扣因子」。
$V_t$ 對未來 30 天平均的影響被均值回歸「折損」了——$\kappa$ 越大（均值回歸越快），$a$ 越小，現在的 $V_t$ 對 VIX 的影響就越弱。

\textbf{意義：}現在你已經知道，VIX futures 的定價問題完全歸結為計算
$F^{\VIX}(t,T) = 100\,\E^\Q_t[\sqrt{aV_T + b}]$——一個只涉及 $V_T$ 的一元函數的期望值。

\section{Step 2：$\E[\sqrt{aV_T+b}]$ 的積分表示（eq 8）}

\subsection{Gaussian 積分的恆等式}

對任意 $x > 0$，由 Gaussian 積分（令 $v = u\sqrt{x}$）：
\[
  \int_0^\infty e^{-xu^2}\,\dd u
  = \frac{1}{\sqrt{x}}\int_0^\infty e^{-v^2}\,\dd v
  = \frac{\sqrt{\pi}}{2\sqrt{x}}
\]

因此：
\begin{equation}
  \sqrt{x} = \frac{2}{\sqrt{\pi}}\int_0^\infty e^{-xu^2}\,\dd u
  \label{eq:sqrt-trick}
\end{equation}

這個等式是 Proposition 1 的技術核心。

\subsection{取期望，得到 Laplace transform 的積分}

把 \eqref{eq:sqrt-trick} 應用到 $x = aV_T + b$，並取期望（交換期望與積分，由 Fubini 定理，在足夠的可積條件下成立）：

\begin{align}
  \E^\Q_t\!\left[\sqrt{aV_T + b}\right]
  &= \frac{2}{\sqrt{\pi}}\int_0^\infty \E^\Q_t\!\left[e^{-(aV_T+b)u^2}\right]\dd u \notag\\
  &= \frac{2}{\sqrt{\pi}}\int_0^\infty e^{-bu^2}\,\E^\Q_t\!\left[e^{-au^2 V_T}\right]\dd u
  \label{eq:prop1-uv}
\end{align}

換積分變數 $w = u^2$（$\dd u = \dd w/(2\sqrt{w})$）：
\begin{align}
  &= \frac{2}{\sqrt{\pi}}\int_0^\infty e^{-bw}\,\E^\Q_t\!\left[e^{-awV_T}\right]\frac{\dd w}{2\sqrt{w}} \notag\\
  &= \frac{1}{\sqrt{\pi}}\int_0^\infty w^{-1/2}\,e^{-bw}\cdot\varphi(-aw, V_t, \tau)\,\dd w
  \label{eq:prop1-w}
\end{align}

其中 $\tau = T - t$，$\varphi(s, V_t, \tau) := \E^\Q_t[e^{sV_T}]$ 是 $V_T$ 的條件 MGF。

\begin{keybox}
\textbf{Proposition 1（eq 8--9）的核心等式：}
\begin{equation}
  F^{\VIX}(t,T)
  = \frac{100}{\sqrt{\pi}}\int_0^\infty w^{-1/2}\,e^{-bw}\cdot\varphi(-aw, V_t, \tau)\,\dd w
  \label{eq:prop1}
\end{equation}
只要 $\varphi(-aw, V_t, \tau)$ 是已知函數，這就是一個一維實值積分，
被積函數在 $w\to\infty$ 時因 $e^{-bw}$ 迅速衰減，$w\to 0$ 時可用 $w^{-1/2}$ 的奇點處理（有限的廣義積分）。
\end{keybox}

\textbf{為什麼是「實值」積分？}

因為 $\varphi(-aw, \cdot)$ 是在實數負軸求 MGF，$V_T \ge 0$ 保證了 $e^{-awV_T} \in [0,1]$，
所以 $\varphi(-aw, \cdot) = \E[e^{-awV_T}] \in (0,1]$ 是實數。

對比標準 Fourier 反演（如 Heston 選擇權定價），那裡需要對\emph{複數}$u$ 計算特徵函數
$\E[e^{iuX}]$，數值上更不穩定，而且需要截斷 truncation error 的控制。

\section{Step 3：Affine MGF 的閉式解（eq 10--11）}

\subsection{問題：求 $\varphi(s, V_t, \tau) = \E^\Q_t[e^{sV_T}]$}

對 SVJJ 模型（僅考慮 $V_t$ 的 SDE，即 eq 4），用 Feynman-Kac 定理。

設 $\varphi(v, \tau) := \E^\Q_t[e^{sV_T}\,|\,V_t=v]$（省略 $s$ 的依賴性）。
由 Feynman-Kac（無折現，$f(V_T) = e^{sV_T}$），$\varphi$ 滿足：
\begin{equation}
  \frac{\partial\varphi}{\partial\tau}
  = \underbrace{\kappa^\Q(\theta^\Q - v)\frac{\partial\varphi}{\partial v}}_{\text{CIR drift}}
  + \underbrace{\frac{1}{2}\sigma_V^2 v\frac{\partial^2\varphi}{\partial v^2}}_{\text{CIR diffusion}}
  + \underbrace{\lambda^\Q\left(\E_{Z^V}[\varphi(v+Z^V,\tau)] - \varphi\right)}_{\text{jump term}}
  \label{eq:phi-pde}
\end{equation}
終端條件：$\varphi(v, 0) = e^{sv}$（即 $\tau = 0 \Rightarrow t = T$）。

\subsection{Affine 假設（Ansatz）}

假設解具有指數仿射形式（eq 10）：
\begin{equation}
  \varphi(v, \tau) = \exp\!\big(C(\tau) + D(\tau)\cdot v + A(\tau)\big)
  \tag{eq 10}
\end{equation}
其中 $C, D, A$ 只是 $\tau$ 的函數，$A$ 含跳躍貢獻（純 CIR 時 $A = 0$）。

\begin{derivbox}
計算 \eqref{eq:phi-pde} 需要的各偏導數（$A$ 先設為 0，只看 diffusion 部分）：
\begin{align*}
  \partial\varphi/\partial\tau &= (\dot C + \dot D\,v)\,\varphi \\
  \partial\varphi/\partial v &= D\,\varphi \\
  \partial^2\varphi/\partial v^2 &= D^2\varphi
\end{align*}
代入 PDE（跳躍項暫略），兩邊除以 $\varphi$：
\[
  \dot C + \dot D\,v
  = \kappa^\Q(\theta^\Q - v)\,D + \tfrac{1}{2}\sigma_V^2 v\,D^2
\]
分離常數項與 $v$ 項：
\begin{align}
  \dot C &= \kappa^\Q\theta^\Q\,D \label{eq:C-ode}\\
  \dot D &= -\kappa^\Q\,D + \tfrac{1}{2}\sigma_V^2\,D^2 \label{eq:D-ode}
\end{align}
初始條件（$\tau = 0$）：$D(0) = s$，$C(0) = 0$。
\end{derivbox}

\begin{keybox}
\textbf{eq 11：Riccati ODE for $D$（含跳躍）：}
\begin{equation}
  \dot D = \tfrac{1}{2}\sigma_V^2\,D^2 - \kappa^\Q\,D + \lambda^\Q\!\left(\psi^V(D) - 1\right)
  \tag{eq 11, 部分}
\end{equation}
其中 $\psi^V(D) = \E[e^{DZ^V}]$ 是方差跳躍幅度 $Z^V$ 的 MGF。
對指數分布 $Z^V\sim\text{Exp}(\mu_V)$：$\psi^V(D) = \mu_V/(\mu_V - D)$（須 $D < \mu_V^{-1}$）。

$A(\tau)$ 只含跳躍對截距的貢獻，$C(\tau)$ 由 \eqref{eq:C-ode} 積分得到。
\end{keybox}

\subsection{純 CIR 的解析解（$\lambda = 0$）}

無跳躍時，Riccati ODE \eqref{eq:D-ode} 是 Bernoulli 方程，令 $h = 1/D$：
$\dot h = \kappa^\Q h - \tfrac{1}{2}\sigma_V^2$，解為：
\[
  h(\tau) = \frac{\kappa^\Q}{\tfrac{1}{2}\sigma_V^2} + \left(\frac{1}{s} - \frac{\kappa^\Q}{\tfrac{1}{2}\sigma_V^2}\right)e^{\kappa^\Q\tau}
\]
整理後（這就是 Heston 模型的特徵函數核心）：
\begin{equation}
  D(\tau) = \frac{s\,e^{-\kappa^\Q\tau}}{1 - \dfrac{\sigma_V^2 s}{2\kappa^\Q}\!\left(1-e^{-\kappa^\Q\tau}\right)}
  \label{eq:D-closed}
\end{equation}
$C(\tau) = \kappa^\Q\theta^\Q\int_0^\tau D(r)\,\dd r$（可積分得到對數形式）。

有跳躍時（SVJJ），Riccati ODE 一般無閉式解，需用 ODE solver 數值積分。

\begin{physicsbox}
\textbf{Affine 結構 = 線性 Hamiltonian 的路徑積分}

式 \eqref{eq:D-ode} 的 Riccati 方程在物理上是 Hamilton-Jacobi 方程（HJE）的特殊情形。

在量子力學中，對哈密頓量為二次型（$H \propto x^2 + p^2$，諧振子）的系統，
作用量 $S[\text{path}]$ 也是路徑的二次型，路徑積分精確等於鞍點展開（Gaussian 積分）——
結果是一個指數函數，其指數是初始/終端條件的有理函數。

CIR 的 SDE 是「有平方根擴散的線性 SDE」，其 generator 是 $v$ 的一次多項式（affine）。
Feynman-Kac 的 PDE 因此也是 $v$ 的一次多項式係數——允許 ansatz $e^{C+Dv}$ 完全分離變數，
Riccati ODE 是 HJE 的 Bernoulli 近親。

這正是為什麼 affine 模型類（Duffie-Pan-Singleton 2000）如此廣泛使用：
線性結構讓 MGF 有解析解，而 MGF 是定價工具箱的通用入口。
\end{physicsbox}

\section{完整公式與數值計算（eq 9）}

\subsection{最終定價公式}

結合 Steps 1--3，VIX 期貨定價公式（eq 8--11 的綜合）：

\begin{keybox}
\begin{equation}
  F^{\VIX}(t,T) = \frac{100}{\sqrt{\pi}}
  \int_0^\infty w^{-1/2}\,e^{-bw}
  \cdot\exp\!\big(C(\tau,-aw) + D(\tau,-aw)\,V_t + A(\tau,-aw)\big)\,\dd w
  \label{eq:full-formula}
\end{equation}
其中 $s = -aw < 0$，$C, D, A$ 來自數值求解 Riccati ODE（eq 11）。
\end{keybox}

\textbf{數值計算流程：}
\begin{enumerate}
  \item 給定模型參數 $(\kappa^\Q, \theta^\Q, \sigma_V, \lambda^\Q, \mu_V, \ldots)$ 和 $V_t$
  \item 計算 $a$, $b$（eq 6）
  \item 對每個 $w$ 值，數值求解 ODE 得 $C(\tau,-aw), D(\tau,-aw), A(\tau,-aw)$
  \item 用 Gaussian quadrature 計算一維積分（因 $e^{-bw}$ 衰減，截斷誤差可控）
\end{enumerate}

\subsection{為什麼不用 Taylor 近似？（Sec 2.3 的主要結論）}

文獻中的近似方法（Lin 2007；Zhang, Shu, Brenner 2010）對 $\sqrt{x}$ 在 $\bar x = \E[\VIX_T^2]$
附近做 Taylor 展開：
\[
  \sqrt{x} \approx \sqrt{\bar x} + \frac{x-\bar x}{2\sqrt{\bar x}}
  - \frac{(x-\bar x)^2}{8\bar x^{3/2}} + \cdots
\]

\textbf{問題：} $V_T$ 的分布在高波動率環境下是\emph{重尾}的（CIR 的穩態是 Gamma 分布，
跳躍讓右尾更厚），展開點 $\bar x$ 離分布質量中心很遠，高階項不可忽略。

Zhu (2011) 圖 1 顯示：在 VIX > 30 或遠月合約下，Taylor 近似誤差超過 1--2 VIX 點。
精確的 \eqref{eq:full-formula} 沒有這個問題。

\section{Riccati ODE 的解析解（純 CIR，$\lambda=0$）}

\subsection*{問題設定}

求解（以 $\tau = T-t$ 為自變數，初始條件 $D(0) = s$）：
\[
  \dot D = \frac{\sigma_V^2}{2}D^2 - \kappa^\Q D,
  \qquad \dot C = \kappa^\Q\theta^\Q D
\]

\subsection*{求解 $D(\tau)$}

這是 Bernoulli 方程。令 $h = 1/D$，則 $\dot h = -\dot D / D^2$，代入：
\[
  -D^2 \dot h = \frac{\sigma_V^2}{2}D^2 - \kappa^\Q D
  \;\Rightarrow\;
  \dot h = -\frac{\sigma_V^2}{2} + \kappa^\Q h
\]
這是線性 ODE，解為：
\[
  h(\tau) = \frac{\sigma_V^2}{2\kappa^\Q} + \left(\frac{1}{s} - \frac{\sigma_V^2}{2\kappa^\Q}\right)e^{\kappa^\Q\tau}
\]
取倒數：
\[
  D(\tau)
  = \frac{1}{h(\tau)}
  = \frac{s\,e^{-\kappa^\Q\tau}}{1 - \dfrac{\sigma_V^2 s}{2\kappa^\Q}(1-e^{-\kappa^\Q\tau})}
\]

以定價問題的角數，$s = -aw \le 0$，分母 $\ge 1$，$D(\tau) \le 0$，不會有奇點。

\subsection*{求解 $C(\tau)$}

\[
  C(\tau) = \kappa^\Q\theta^\Q\int_0^\tau D(r)\,\dd r
\]

代入 $D(r)$ 並積分（令 $\gamma = \kappa^\Q$，$\beta = \sigma_V^2 s / (2\kappa^\Q)$）：
\begin{align*}
  \int_0^\tau \frac{s\,e^{-\gamma r}}{1 - \beta(1-e^{-\gamma r})}\,\dd r
  &= \frac{s}{\gamma}\,\frac{1}{1/\beta - 1}
    \ln\!\left[\frac{1-\beta(1-e^{-\gamma\tau})}{1-\beta}\right]
    + \cdots
\end{align*}
完整表達式見論文 eq 11（有多種等價寫法）。
對 Heston 選擇權定價的讀者：這是特徵函數 $\E[e^{iuX_T}]$ 裡 $C$ 項的實部版本。
